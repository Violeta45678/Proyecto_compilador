// Funciones, parámetros, llamadas anidadas y recursión
string saludo = "Hola mundo";

int max(int a, int b) {
    if (a > b) return a;
    return b;
}

float promedio(float x, float y, float z) {
    return (x + y + z) / 3.0;
}

int fib(int n) {
    if (n <= 1) return n;
    return fib(n - 1) + fib(n - 2);
}

void imprimirTodo() {
    imprimir(saludo);
    imprimir(max(fib(5), max(3, 4)));
    imprimir(promedio(1.5, 2.0, max(1, 2)));
}
